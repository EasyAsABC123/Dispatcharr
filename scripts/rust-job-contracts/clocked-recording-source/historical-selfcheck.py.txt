import argparse,ast,hashlib,json
from pathlib import Path
def sha(path):return hashlib.sha256(Path(path).read_bytes()).hexdigest()
def need(value,label):
 if value is not True:raise ValueError(label)
def verify(root,external=True):
 root=Path(root);inputs=json.loads((root/'proposed-execution-inputs.json').read_text());tools=root/'tools'
 tree=ast.parse((tools/'run_source.py').read_text());required=next(ast.literal_eval(n.value)for n in tree.body if isinstance(n,ast.Assign)and any(isinstance(t,ast.Name)and t.id=='REQUIRED'for t in n.targets))
 need(len(required)==17 and set(inputs['tools_sha256'])==required,'ExactHostRequiredSet17')

 need(inputs['authorized']is False,'ProposedNotAuthorized')
 need(inputs['pinned_source_proof_sha256']==sha(root/'pinned-source-proof.json')and inputs['case_plan_sha256']==sha(root/'case-plan.json'),'ExactProofAndPlan')

 original=json.loads((root/'pinned-source-proof.json').read_text())
 for name,row in original['modules'].items():need(sha(root/'original'/name)==row['sha256'],'ExactOriginalModule:'+name)
 for path in tools.glob('*.py'):ast.parse(path.read_text(),filename=str(path))
 output=Path(inputs['output']);need(not output.exists(),'FreshExactOutput')

  for key in ('Original_image_inspect','tooling_environment_script','compose_executable'):
   obj=inputs[key];need(sha(obj['path'])==obj['sha256'],'ExternalReceipt:'+key)
 return {'passed':True,'Source_app_executed':False,'Native_app_executed':False,'tools':len(required),'Original_modules':len(original['modules']),'support_files':len(full['files']),'fresh_output':str(output),'execution_authorized':False,'actual_container_image_inspect_executed_here':False}
if __name__=='__main__':
